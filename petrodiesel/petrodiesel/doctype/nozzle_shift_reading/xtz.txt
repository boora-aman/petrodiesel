# Create all workspaces
for ws_data in workspaces_data:
    ws_label = ws_data["label"]
    
    # Delete if exists
    if frappe.db.exists("Workspace", ws_label):
        frappe.delete_doc("Workspace", ws_label, force=1)
        print(f"🗑️  Deleted old: {ws_label}")
    
    # Create new workspace
    workspace = frappe.get_doc({
        "doctype": "Workspace",
        "label": ws_label,
        "title": ws_data["title"],
        "icon": ws_data["icon"],
        "indicator_color": ws_data["indicator_color"],
        "module": "Petrodiesel",
        "public": 1,
        "is_hidden": 0
    })
    
    # Add shortcuts
    for shortcut in ws_data["shortcuts"]:
        workspace.append("shortcuts", shortcut)
    
    # Add links
    for link in ws_data["links"]:
        workspace.append("links", {
            "type": "Link",
            "link_type": link["link_type"],
            "link_to": link["link_to"],
            "label": link["label"],
            "is_query_report": 1 if link["link_type"] == "Report" else 0,
            "onboard": 0,
            "hidden": 0
        })
    
    workspace.insert(ignore_permissions=True)
    print(f"✅ Created: {ws_label}")

frappe.db.commit()
